# Notes:  IamInstanceProfileName is the IAM role attached to an EC2 instance.
#         c7i.large   2 CPU
#         c7i.2xlarge 8 CPU
#         c7i.8xlarge 32 CPU  
#         c7i.16xlarge 64 CPU
#         c7i.24xlarge 96 CPU

aws cloudformation deploy \
  --stack-name valley-scanner-box \
  --template-file EC2_4_machines_test.yaml \
  --capabilities CAPABILITY_NAMED_IAM \
  --parameter-overrides \
    AmiId=ami-0dfe9aa3d33b19244 \
    InstanceType=c7i.16xlarge  \
    SubnetId=subnet-0fbcb444dd2b8da11 \
    SecurityGroupId=sg-0a4c92ac491102653 \
    IamInstanceProfileName=z-batch-runner \
    NamePrefix=z-valley-worker

  --table-name valley-scanner-dev-jobs \
  --index-name status-index \
