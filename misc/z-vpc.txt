# ---------- config ----------
REGION=us-east-1
VPC_CIDR=10.20.0.0/16
AZ1=us-east-1a
AZ2=us-east-1b
SUBNET1_CIDR=10.20.1.0/24
SUBNET2_CIDR=10.20.2.0/24
TAGS='ResourceType=vpc,Tags=[{Key=Name,Value=z-VPC}]'

# ---------- VPC ----------
VPC_ID=$(aws ec2 create-vpc --cidr-block $VPC_CIDR \
  --tag-specifications "$TAGS" \
  --query 'Vpc.VpcId' --output text --region $REGION)

# enable DNS hostnames (good hygiene; helps SSM etc. later)
aws ec2 modify-vpc-attribute --vpc-id $VPC_ID --enable-dns-hostnames --region $REGION

# ---------- PRIVATE subnets (no IGW, no NAT) ----------
SUBNET1_ID=$(aws ec2 create-subnet --vpc-id $VPC_ID --availability-zone $AZ1 \
  --cidr-block $SUBNET1_CIDR \
  --tag-specifications "ResourceType=subnet,Tags=[{Key=Name,Value=z-private-$AZ1}]" \
  --query 'Subnet.SubnetId' --output text --region $REGION)

SUBNET2_ID=$(aws ec2 create-subnet --vpc-id $VPC_ID --availability-zone $AZ2 \
  --cidr-block $SUBNET2_CIDR \
  --tag-specifications "ResourceType=subnet,Tags=[{Key=Name,Value=z-private-$AZ2}]" \
  --query 'Subnet.SubnetId' --output text --region $REGION)

# ---------- Route tables for the private subnets ----------
RTB1=$(aws ec2 create-route-table --vpc-id $VPC_ID \
  --tag-specifications "ResourceType=route-table,Tags=[{Key=Name,Value=z-rtb-$AZ1}]" \
  --query 'RouteTable.RouteTableId' --output text --region $REGION)
RTB2=$(aws ec2 create-route-table --vpc-id $VPC_ID \
  --tag-specifications "ResourceType=route-table,Tags=[{Key=Name,Value=z-rtb-$AZ2}]" \
  --query 'RouteTable.RouteTableId' --output text --region $REGION)

aws ec2 associate-route-table --route-table-id $RTB1 --subnet-id $SUBNET1_ID --region $REGION
aws ec2 associate-route-table --route-table-id $RTB2 --subnet-id $SUBNET2_ID --region $REGION

# (No IGW/NAT routes—these remain PRIVATE)

# ---------- S3 Gateway VPC Endpoint (private path to S3) ----------
aws ec2 create-vpc-endpoint \
  --vpc-endpoint-type Gateway \
  --vpc-id $VPC_ID \
  --service-name com.amazonaws.$REGION.s3 \
  --route-table-ids $RTB1 $RTB2 \
  --tag-specifications "ResourceType=vpc-endpoint,Tags=[{Key=Name,Value=z-s3-endpoint}]" \
  --region $REGION

# ---------- Security Group for compute ----------
SG_ID=$(aws ec2 create-security-group \
  --vpc-id $VPC_ID \
  --group-name z-compute-sg \
  --description "No inbound; egress only to S3 via endpoint or default" \
  --tag-specifications "ResourceType=security-group,Tags=[{Key=Name,Value=z-compute-sg}]" \
  --query 'GroupId' --output text --region $REGION)

# Inbound: none (default is empty) — good.
# Egress: default allows all. That's safe in a private subnet (no IGW/NAT).
# Optional: restrict egress to S3 only using the managed prefix list:
PL_S3=$(aws ec2 describe-managed-prefix-lists \
  --filters Name=prefix-list-name,Values=com.amazonaws.$REGION.s3 \
  --query 'PrefixLists[0].PrefixListId' --output text --region $REGION)

# Revoke the default all-egress (optional hardening)
aws ec2 revoke-security-group-egress \
  --group-id $SG_ID \
  --ip-permissions 'IpProtocol=-1,IpRanges=[{CidrIp=0.0.0.0/0}]' --region $REGION || true

# Add just HTTPS to S3 prefix list
aws ec2 authorize-security-group-egress \
  --group-id $SG_ID \
  --ip-permissions "IpProtocol=tcp,FromPort=443,ToPort=443,PrefixListIds=[{PrefixListId=$PL_S3}]" \
  --region $REGION

echo "VPC: $VPC_ID"
echo "Private subnets: $SUBNET1_ID, $SUBNET2_ID"
echo "Compute SG: $SG_ID"
